int[] list = { 64, 34, 25, 12, 22, 11, 90 };

int[] lst = Booble(list);

static int[] Booble(int[] list)
{
    int[] lst = list;
    for (int i = 0; i < list.Length - 1; i++)
    {
        bool res = true;
            if (lst[i] > lst[i + 1])
            {
            int tes = lst[i];
            lst[i] = lst[i + 1];
            lst[i + 1] = tes;
            res = false;
            }
            else
            lst[i] = lst[i];
        if (!res)
            i = -1;
    }
    return lst;
}
