using System;
Console.WriteLine("Задайте кол-во роботов на производство:");
int count = int.Parse(Console.ReadLine());
string[] robos = new string[count];
int rob = 0;
while (rob < count)
{
    Console.WriteLine("Сборка запускается");
    Console.WriteLine();
    Thread.Sleep(400);
    bool pass = CreateProcc();
    if (pass)
    {
        robos[rob] = $"Робот {rob + 1} - Готовый";
        Console.WriteLine();
        Thread.Sleep(2000);
    }
    else
    {
        robos[rob] = $"Робот {rob + 1} - Непригоден";
        Console.WriteLine();
        Thread.Sleep(2000);
    }
    Console.Clear();
    rob++;
}

for (int i = 0; i < count; i++)
{
    Console.WriteLine(robos[i]);
}

static bool CreateProcc()
{
    string stop = "Остановливаем конвейер.";
    Console.WriteLine("Шаг 1 - Установка корпуса");
    if (!Checking("корпус"))
    {
        Console.WriteLine($"Корпус неисправен. {stop}");
        return false;
    }
    Console.WriteLine("Шаг 2 - Установка процессора");
    if (!Checking("процессор"))
    {
        Console.WriteLine($"Процессор неисправен. {stop}");
        return false;
    }
    Console.WriteLine("Шаг 3 - Установка памяти");
    if (!Checking("память"))
    {
        Console.WriteLine($"Память неисправен. {stop}");
        return false;
    }
    Console.WriteLine("Шаг 4 - Установка датчиков");
    if (!Checking("датчиков"))
    {
        Console.WriteLine($"Датчик неисправен. {stop}");
        return false;
    }
    Console.WriteLine("Робот готов!");
    return true;
}

static bool Checking(string component)
{
    Console.Write($"Проверка {component} на качество == ");
    Thread.Sleep(800);
    Random rand = new Random();
    bool lok = rand.Next(100) > 15;
    if (lok)
        Console.WriteLine("Положительный");
    else
        Console.WriteLine("Отрицательный");
    Console.WriteLine();
    return lok;
}
